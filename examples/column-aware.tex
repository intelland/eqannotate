\section*{Column-aware bounds}
\eqannotatecolortheme{colorful}
\eqannotatecalloutstyle{arrow}
\begin{annotatedequation}
\eqmark[blue]{left}{\alpha}
+\beta+\gamma+\delta+\epsilon+\zeta+
\eqmark[red]{right}{\eta}
\eqannotate[prefer=above]{left}{A deliberately long label near the left edge}
\eqannotate[prefer=below]{right}{Another deliberately long label near the right edge}
\end{annotatedequation}

\vspace{1em}
\noindent The labels should remain inside this column even though their ideal centers are near the edge terms.
